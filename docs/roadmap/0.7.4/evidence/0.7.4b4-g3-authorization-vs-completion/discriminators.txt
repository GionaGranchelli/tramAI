class G3TransientProbeTest {
    private val runId = "run-g3-probe"
    private val correlationId = "corr-g3-probe"

    @Test
    fun `an ALLOW record survives a delegate that throws after authorization`() {
        val delegate =
            object : ToolCallback {
                override fun getToolDefinition(): ToolDefinition =
                    ToolDefinition
                        .builder()
                        .name("ledger_append")
                        .description("throws after being authorized")
                        .inputSchema("""{"type":"object","properties":{"value":{"type":"string"}}}""")
                        .build()

                override fun call(toolInput: String): String = error("delegate failed after authorization")
            }

        val boundary =
            GovernedToolCallback(
                delegate = delegate,
                engine =
                    DefaultPolicyEngine(
                        PolicyConfiguration(
                            allowedTools = setOf("ledger_append"),
                            allowedPermissions = setOf("ledger.append"),
                        ),
                    ),
                auditEngine = AuditEngine(InMemoryAuditStore()),
                context = boundaryContext(),
            )

        val failure = runCatching { boundary.call("""{"value":"boom"}""") }.exceptionOrNull()
        val evidence = boundary.evidence()

        println("PROBE delegate outcome: ${failure?.let { it::class.simpleName + ": " + it.message }}")
        println("PROBE recorded events: ${boundary.events().map { it.decision }}")
        println("PROBE evidence records: ${evidence.map { it.decision.kind to it.decision.reasonCode }}")

        assertTrue(failure != null, "the delegate's failure reaches the caller")
        assertEquals(1, evidence.size, "exactly one decision record was produced")
        assertEquals("ALLOW", evidence.single().decision.kind, "the record attests the authorization")
        assertTrue(boundary.events().single().decision == "ALLOW", "the audit decision is ALLOW")
    }

    private fun boundaryContext(): BoundaryContext =
        BoundaryContext(
            identity =
                GovernedRunIdentity(
                    deployment =
                        WorkloadDeploymentIdentity(
                            workloadId = WorkloadId("workload-xr1"),
                            configuration =
                                WorkloadConfigurationIdentity(
                                    id = ConfigurationId("configuration-xr1"),
                                    version = ConfigurationVersion("v1"),
                                ),
                            environmentId = EnvironmentId("environment-local"),
                            deploymentId = DeploymentId("deployment-xr1-1"),
                        ),
                    runId = RunId(runId),
                ),
            security =
                ToolSecurityMetadata(
                    permission = "ledger.append",
                    risk = RiskLevel.LOW,
                    approval = ApprovalMode.AUTO,
                    managedNetworkEgress = ManagedNetworkEgress.ALLOW,
                    audit = AuditDetail.DECISION_ONLY,
                ),
            policyVersion = "policy-xr1-v1",
            actorId = "spring-ai-workload",
            correlationId = correlationId,
        )
}
